cmake_minimum_required(VERSION 3.20)
project(AlignITK LANGUAGES C CXX)
find_package(ITK 5.4 REQUIRED COMPONENTS ITKCommon ITKTransform ITKRegistrationMethodsv4 ITKMetricsv4 ITKOptimizersv4 ITKBiasCorrection ITKImageIntensity ITKThresholding ITKImageGrid ITKSmoothing)
include(${ITK_USE_FILE})
# SDK C headers must not precede libc++ standard headers.
get_directory_property(align_includes INCLUDE_DIRECTORIES)
list(FILTER align_includes EXCLUDE REGEX "SDKs/.*/usr/include$")
set_property(DIRECTORY PROPERTY INCLUDE_DIRECTORIES "${align_includes}")
add_library(align-itk SHARED align_itk.cpp)
target_compile_features(align-itk PRIVATE cxx_std_17)
target_include_directories(align-itk PRIVATE vendor)
target_link_libraries(align-itk PRIVATE ${ITK_LIBRARIES})
set_target_properties(align-itk PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
 target_compile_options(align-itk PRIVATE /O2)
else()
 target_compile_options(align-itk PRIVATE -O3 -fno-fast-math)
endif()
