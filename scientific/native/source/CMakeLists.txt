cmake_minimum_required(VERSION 3.18)
project(JIP LANGUAGES C CXX)
include(CTest)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "JIP requires a 64-bit compiler/toolchain. 32-bit builds are unsupported.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
add_compile_definitions(_FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE=1)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fcommon -Wno-implicit-int -Wno-implicit-function-declaration)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wno-deprecated-non-prototype -Wno-incompatible-function-pointer-types)
endif()
if(MINGW)
  add_compile_definitions(__USE_MINGW_ANSI_STDIO=1)
endif()
find_package(ZLIB REQUIRED)
find_path(FFTW_INCLUDE_DIR fftw3.h PATHS /opt/homebrew/include /usr/local/include REQUIRED)
find_library(FFTW_LIBRARY NAMES fftw3 PATHS /opt/homebrew/lib /usr/local/lib REQUIRED)
set(JIP_INCLUDES include include/recipes jip-align jip-display/common jip-glm jip-rem jip-srtm lib/lmmin nifticlib-2.0.0/include nifticlib-2.0.0/niftilib nifticlib-2.0.0/znzlib)
include_directories(${JIP_INCLUDES} ${FFTW_INCLUDE_DIR})
add_library(jip-vendor STATIC lib/lmmin/lmmin.c lib/lmmin/lm_eval.c nifticlib-2.0.0/niftilib/nifti1_io.c nifticlib-2.0.0/znzlib/znzlib.c)
target_compile_definitions(jip-vendor PUBLIC HAVE_ZLIB=1)
target_link_libraries(jip-vendor PUBLIC ZLIB::ZLIB)
add_library(jip-core STATIC lib/util.c lib/io.c lib/jip-fit.c jip-align/common-align.c)
target_link_libraries(jip-core PUBLIC jip-vendor ${FFTW_LIBRARY})
if(NOT WIN32)
  target_link_libraries(jip-core PUBLIC m)
endif()
function(jip_strict target)
  if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(${target} PRIVATE -Werror=shorten-64-to-32 -Werror=format -Werror=incompatible-pointer-types -Werror=implicit-function-declaration -Werror=implicit-int)
  elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(${target} PRIVATE -Werror=format -Werror=incompatible-pointer-types -Werror=implicit-function-declaration -Werror=implicit-int)
  endif()
endfunction()
jip_strict(jip-core)
add_executable(jip jip/jip-front.c jip/jip-align.c jip/jip-average.c jip/jip-macros.c jip/jip-magcor.c jip/jip-math.c jip/jip-transfilt.c jip/jip-util.c jip/jip-ovl.c jip/jip-io.c jip/jip-reorder.c jip/jip-stats.c jip/jip-glm.c jip/jip-maps.c jip-glm/glm.c)
add_executable(jip-glm jip-glm/fmri-glm.c jip-glm/glm.c)
add_executable(jip-rem jip-rem/fmri-rem.c jip-rem/rem.c)
add_executable(jip-srtm jip-srtm/srtm-glm.c jip-srtm/srtm.c)
foreach(tool jip jip-glm jip-rem jip-srtm)
  target_link_libraries(${tool} PRIVATE jip-core)
  jip_strict(${tool})
endforeach()
option(JIP_BUILD_X11 "Build the X11 aligner and viewer" ON)
option(JIP_BUILD_ENGINE "Build the standalone registration engine" ON)
option(JIP_TEST_X11 "Run opt-in private-display GUI restore/fitting tests (Python numpy/nibabel, Xvfb, Xtst required)" OFF)
if(JIP_BUILD_ENGINE)
  add_subdirectory(engine)
endif()
if(JIP_BUILD_X11)
  find_package(X11 REQUIRED)
  include_directories(${X11_INCLUDE_DIR})
  set(DISPLAY_COMMON jip-display/common/common-display.c jip-display/common/common-X11.c jip-display/common/common-actions.c jip-display/common/common-io.c)
  add_executable(jip-align jip-align/align-Xwindows.c jip-align/align-actions.c jip-align/align-util.c jip-align/align-io.c jip-align/align-affine.c jip-align/align-nonlinear.c ${DISPLAY_COMMON})
  add_executable(jip-display jip-display/display-Xwindows.c jip-display/display-io.c jip-display/display-actions.c jip-display/display-util.c jip-display/display-time-model.c jip-display/display-glm.c jip-display/display-rem.c jip-display/display-srtm.c jip-glm/glm.c jip-rem/rem.c jip-srtm/srtm.c ${DISPLAY_COMMON})
  foreach(tool jip-align jip-display)
    target_link_libraries(${tool} PRIVATE jip-core ${X11_LIBRARIES})
    jip_strict(${tool})
  endforeach()
  if(BUILD_TESTING AND JIP_TEST_X11)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    find_program(JIP_XVFB_EXECUTABLE NAMES Xvfb PATHS /opt/X11/bin REQUIRED)
    file(MAKE_DIRECTORY ${CMAKE_BINARY_DIR}/gui-regression)
    add_test(NAME gui-restore-and-fit COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/../tests/test_gui_restore.py
      --binary $<TARGET_FILE:jip-align> --xvfb ${JIP_XVFB_EXECUTABLE}
      --output ${CMAKE_BINARY_DIR}/gui-regression)
    set_tests_properties(gui-restore-and-fit PROPERTIES LABELS "gui;x11;restore;registration" TIMEOUT 120)
  endif()
endif()
